PART 3 — CRUD: COURSES COLLECTION
------------------------------------------------------------

INSERT courses:
db.courses.insertMany([
    { _id: "CS101", courseName: "Computer Science",    department: "Computer Applications",  credits: 4 },
    { _id: "DB101", courseName: "Database Management", department: "Computer Applications",  credits: 3 },
    { _id: "NW101", courseName: "Computer Networks",   department: "Information Technology", credits: 4 },
    { _id: "DS101", courseName: "Data Science",        department: "Computer Applications",  credits: 4 }
])

READ all courses:
db.courses.find()
db.courses.find().pretty()

READ one course by ID:
db.courses.findOne({ _id: "CS101" })

UPDATE course credits:
db.courses.updateOne(
    { _id: "CS101" },
    { $set: { credits: 5 } }
)

UPSERT — add DS101 if not present:
db.courses.updateOne(
    { _id: "DS101" },
    {
        $set: {
            courseName: "Data Science",
            department: "Computer Applications",
            credits: 4
        }
    },
    { upsert: true }
)

DELETE a course:
db.courses.deleteOne({ _id: "DS101" })

PART 6 — AGGREGATION: COURSE PERFORMANCE REPORT
------------------------------------------------------------

db.students.aggregate([
    {
        $group: {
            _id: "$courseId",
            totalStudents: { $sum: 1 },
            averageMarks: { $avg: "$marks" }
        }
    },
    {
        $lookup: {
            from: "courses",
            localField: "_id",
            foreignField: "_id",
            as: "course"
        }
    },
    { $unwind: "$course" },
    {
        $project: {
            _id: 0,
            courseId: "$_id",
            courseName: "$course.courseName",
            totalStudents: 1,
            averageMarks: { $round: ["$averageMarks", 2] }
        }
    }
])

  { totalStudents: 2, courseId: 'NW101', courseName: 'Computer Networks',   averageMarks: 85   },
  { totalStudents: 3, courseId: 'CS101', courseName: 'Computer Science',    averageMarks: 85   },
  { totalStudents: 2, courseId: 'DB101', courseName: 'Database Management', averageMarks: 91.5 }
]
